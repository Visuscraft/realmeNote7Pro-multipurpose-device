# Background services and OS features disabled by deploy/debloat.sh to reduce
# CPU, RAM and battery usage on a device that runs unattended.
#
# Format: one directive per line, "#" starts a comment.
#
#   setting <namespace> <key> <value>   write a value with "settings put"
#                                       (namespace is global, system or secure)
#   standby <package>                   force the package into the "restricted"
#                                       app standby bucket
#
# The previous value of every setting is recorded in
# $DEVICE_HOME/state/debloat-settings.bak so "--restore" can put it back.

# --- Scanning that keeps the radios and the CPU busy -------------------------
setting global wifi_scan_always_enabled 0
setting global ble_scan_always_enabled 0
setting global network_recommendations_enabled 0
setting global wifi_networks_available_notification_on 0

# --- Telemetry, usage reporting and crash uploads ----------------------------
setting global send_action_app_error 0
setting secure send_action_app_error 0

# --- Animations: pure GPU/CPU cost on a headless device ----------------------
setting global window_animation_scale 0
setting global transition_animation_scale 0
setting global animator_duration_scale 0

# --- Assistant, hotword detection and always-on displays ---------------------
setting secure voice_interaction_service ""
setting secure assistant ""
setting secure doze_always_on 0
setting system screen_brightness_mode 0

# --- Aggressive power management for the whole system ------------------------
setting global adaptive_battery_management_enabled 1
setting global app_standby_enabled 1
setting global forced_app_standby_enabled 1
setting global low_power_sticky 0

# --- Heavy stock components pushed into the restricted standby bucket --------
standby com.google.android.gms
standby com.android.vending
standby com.heytap.mcs
standby com.coloros.push
